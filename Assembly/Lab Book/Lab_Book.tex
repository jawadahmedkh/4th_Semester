\section{\color{magenta}\underline{Hello World in Assembly}}
\subsection{Description}
This is a simple assembly language program that displays the message "Hello, from Assembly!" on the screen.

\subsection{Code}
\begin{tcolorbox}[colback=white,colframe=black,title=Hello World Program]
\begin{verbatim}
.686
INCLUDE Irvine32.inc 
.data
    msg BYTE "Hello, from Assembly!",0
.code
    Main PROC
        lea edx, msg
        Call Clrscr
        Call WriteString
        EXIT
    Main ENDP
    END Main
\end{verbatim}
\end{tcolorbox}

\subsection{Output}
\begin{tcolorbox}[colback=white,colframe=black,title=Output]
\begin{verbatim}
Hello, from Assembly!
\end{verbatim}
\end{tcolorbox}
